F = \overline{x \cdot x}

\begin{aligned}
F &= \overline{x \cdot x} \\

\end{aligned}
$$

Hvilket er definisjonen av NOT-porten.

\subsection{AND}
\begin{equation}
\label{eq:and-av-nand}
F = \overline{\overline{xy} \cdot \overline{xy}}
\end{equation}

AND konstrueres av NAND-porter som illustrert i \cref{fig:and-av-nand}. Dette gir det algebraiske utrykket i \cref{eq:and-av-nand}. Utrykket kan forenkles slik:

$$
\begin{aligned}
F &= \overline{\overline{xy} \cdot \overline{xy}} \\
&= \overline{\overline{xy}} \text{ (Idempotens)} \\
&= xy \text{ (Dobbel invertering)}
\end{aligned}
$$

Hvilket er definisjonen av AND-porten.

\subsection{OR}

\begin{equation}
\label{eq:or-av-nand}
F = \overline{\overline{xx} \cdot \overline{yy}}
\end{equation}

OR konstrueres av NAND-porter som illustrert i \cref{fig:or-av-nand}. Dette gir det algebraiske utrykket i \cref{eq:or-av-nand}. Utrykket kan forenkles slik:

$$
\begin{aligned}
F &= \overline{\overline{xx} \cdot \overline{yy}} = \overline{\overline{x} \cdot \overline{y}} \\
&= \overline{\overline{x}} + \overline{\overline{y}} \text{ (DeMorgan)} \\
&= x + y \text{ (Dobbel invertering)}
\end{aligned}
$$

Hvilket er definisjonen av OR-porten.

\subsection{XOR}

\begin{equation}
\label{eq:xor-av-nand}
F = \overline{\overline{\overline{xy} \cdot x} \cdot \overline{\overline{xy} \cdot y} }
\end{equation}

XOR konstrueres av NAND-porter som illustrert i \cref{fig:xor-av-nand}. Dette gir det algebraiske utrykket i \cref{eq:xor-av-nand}. Utrykket kan forenkles slik:

$$
\begin{aligned}
F &= \overline{\overline{\overline{xy} \cdot x} \cdot \overline{\overline{xy} \cdot y} } \\
&= (\overline{\overline{\overline{xy} \cdot x}}) + (\overline{\overline{\overline{xy} \cdot y}}) \text{ (DeMorgan)} \\
&= (\overline{xy} \cdot x) + (\overline{xy} \cdot y) \text{ (Dobbel invertering)} \\
&= \overline{xy}(x + y) \text{ (Faktorisering)} \\
&= (\overline{x} + \overline{y})(x + y) \text{ (DeMorgan)} \\
&= \overline{x}x + \overline{y}x + \overline{x}y + \overline{y}y \text{ (Distribusjon)} \\
&= 0 + \overline{y}x + \overline{x}y + 0 \\
&= x\overline{y} + \overline{x}y
\end{aligned}
$$

Hvilket er definisjonen av XOR-porten:

$$
x\overline{y} + \overline{x}y = x \oplus y
$$

\subsection{XNOR}

\begin{equation}
\label{eq:xnor-av-nand}
F = \overline{\overline{\overline{xx} \cdot \overline{yy}} \cdot \overline{xy}}
\end{equation}

XOR konstrueres av NAND-porter som illustrert i \cref{fig:xnor-av-nand}. Dette gir det algebraiske utrykket i \cref{eq:xnor-av-nand}. Utrykket kan forenkles slik:

$$
\begin{aligned}
F &= \overline{\overline{\overline{xx} \cdot \overline{yy}} \cdot \overline{xy}} \\
&= \overline{\overline{\overline{x} \cdot \overline{y}} \cdot \overline{xy}} \\
&= (\overline{\overline{\overline{x} \cdot \overline{y}}}) + (\overline{\overline{x \cdot y}}) \text{ (DeMorgan)} \\
&= (\overline{x} \cdot \overline{y}) + (x \cdot y) \text{ (Dobbel invertering)} \\
&= xy + \overline{x}\overline{y}
\end{aligned}
$$

Hvilket er definisjonen av XNOR-porten:

$$
xy + \overline{x}\overline{y} = x \odot y
$$

\subsection{Realisering av XNOR ved hjelp av 74F00}
XNOR-porten kan ikke realiseres med en enkelt 74F00-IC, fordi XNOR-porten krever 5 NAND-porter for å realisere. 74F00 inneholder kun 4 NAND-porter. Den kan realiseres med to 74F00-ICer, eller ved hjelp av en 74F00 og en lignende IC med minst 2 NOT-porter. Dette er fordi 2 av NAND-portene kan erstattes med NOT-porter.
